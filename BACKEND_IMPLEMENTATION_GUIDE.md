
# Facebook Ad Reporting CRM Tool - Backend Implementation Guide

## 🎯 Overview
Internal CRM tool for processing Facebook ad account data, generating product-wise reports with smart campaign categorization, and tracking delivery status.

## 🔧 Core Workflow

### 1. Setup Tab - Ad Account Sync
- **UI**: Single "Sync Ad Accounts" button (no product keywords input needed)
- **Process**: Fetch all ad accounts from Facebook Business Manager
- **Loading**: 2-5 minutes allowed (Facebook API throttling)

### 2. Smart Report Generation
- **Trigger**: CRM clicks "Generate Report" per account
- **Smart Detection**: Auto-detect product keywords from campaign names
- **Confirmation**: Show modal if multiple products detected

### 3. Product Categorization & Comments
- **Per-product comments**: CRM adds observations for each detected product
- **Fallback**: Option to combine all campaigns if no grouping desired

### 4. PDF Generation & Delivery
- **Watermarked PDF**: "Generated by DigitalInclined | Internal CRM Report"
- **WhatsApp snippet**: Auto-generated message for client communication
- **Delivery tracking**: Mark as sent, update timestamps

## 🧠 Smart Product Detection Algorithm

```javascript
function detectProductKeywords(campaignNames) {
  // Words to ignore in campaign names
  const ignoreWords = [
    'lookalike', 'retarget', 'remarketing', 'broad', 
    'repurchase', 'tof', 'bof', 'top', 'funnel', 
    'middle', 'roas', 'awareness', 'conversion',
    'campaign', 'ad', 'test', 'new', 'old'
  ];
  
  const keywordCandidates = {};
  
  campaignNames.forEach(name => {
    // Split by common delimiters: hyphen, underscore, space
    const parts = name.split(/[-_\s]+/);
    
    parts.forEach(part => {
      const word = part.toLowerCase().trim();
      
      // Filter criteria for product keywords
      if (
        word.length >= 2 && word.length <= 6 && // Short codes
        !ignoreWords.includes(word) && // Not generic terms
        /^[a-z0-9]+$/i.test(word) && // Alphanumeric only
        !isNaN(word) === false // Not purely numeric
      ) {
        const upperWord = word.toUpperCase();
        keywordCandidates[upperWord] = (keywordCandidates[upperWord] || 0) + 1;
      }
    });
  });
  
  // Return keywords that appear in 2+ campaigns
  return Object.entries(keywordCandidates)
    .filter(([_, count]) => count >= 2)
    .map(([word]) => word)
    .slice(0, 5); // Max 5 products for UI clarity
}

// Example usage:
const campaigns = [
  "OSC-May-Launch",
  "OSC-Retargeting-June", 
  "ABC-BOF-Test",
  "ABC-Conversion-Campaign"
];

detectProductKeywords(campaigns); // Returns: ["OSC", "ABC"]
```

### Rate Limiting Strategy
```javascript
const rateLimiter = {
  maxCallsPerHour: 200, // Facebook limit
  delayBetweenCalls: 500, // 500ms delay
  retryAttempts: 3,
  exponentialBackoff: true
};

async function makeApiCall(url, params) {
  for (let attempt = 1; attempt <= rateLimiter.retryAttempts; attempt++) {
    try {
      await delay(rateLimiter.delayBetweenCalls * attempt);
      const response = await fetch(url, params);
      
      if (response.status === 429) { // Rate limited
        const waitTime = Math.pow(2, attempt) * 1000; // Exponential backoff
        await delay(waitTime);
        continue;
      }
      
      return response.json();
    } catch (error) {
      if (attempt === rateLimiter.retryAttempts) throw error;
    }
  }
}
```

## 📋 Product Metrics Calculation

```javascript
async function calculateProductMetrics(adAccountId, dateRange, productKeywords) {
  const campaigns = await fetchCampaigns(adAccountId);
  const insights = await fetchInsights(adAccountId, dateRange);
  
  // Group campaigns by detected products
  const productGroups = {};
  
  campaigns.forEach(campaign => {
    const detectedProduct = detectProductFromName(campaign.name, productKeywords);
    const productKey = detectedProduct || 'Uncategorized';
    
    if (!productGroups[productKey]) {
      productGroups[productKey] = [];
    }
    productGroups[productKey].push(campaign.id);
  });
  
  // Calculate metrics per product
  const productMetrics = await Promise.all(
    Object.entries(productGroups).map(async ([productName, campaignIds]) => {
      const productInsights = insights.filter(i => 
        campaignIds.includes(i.campaign_id)
      );
      
      const totalSpend = productInsights.reduce((sum, i) => sum + parseFloat(i.spend || 0), 0);
      const totalImpressions = productInsights.reduce((sum, i) => sum + parseInt(i.impressions || 0), 0);
      const totalClicks = productInsights.reduce((sum, i) => sum + parseInt(i.clicks || 0), 0);
      
      // Facebook leads from lead ads
      const fbLeads = await fetchLeadsForCampaigns(campaignIds, dateRange);
      
      // Validated leads from external source (Google Sheets/webhook)
      const validatedLeads = await fetchValidatedLeads(campaignIds, dateRange);
      
      return {
        productName,
        spend: Math.round(totalSpend),
        fbLeads: fbLeads.length,
        validatedLeads: validatedLeads.length,
        cplFb: fbLeads.length > 0 ? Math.round((totalSpend / fbLeads.length) * 100) / 100 : 0,
        cplValidated: validatedLeads.length > 0 ? Math.round((totalSpend / validatedLeads.length) * 100) / 100 : 0,
        cpm: totalImpressions > 0 ? Math.round((totalSpend / totalImpressions * 1000) * 100) / 100 : 0,
        ctr: totalImpressions > 0 ? Math.round((totalClicks / totalImpressions * 100) * 100) / 100 : 0,
        comments: '' // To be filled by CRM
      };
    })
  );
  
  return productMetrics;
}
```

### Google Sheets (Lead Validation)
```javascript
const { GoogleSpreadsheet } = require('google-spreadsheet');

async function fetchValidatedLeads(campaignIds, dateRange) {
  const doc = new GoogleSpreadsheet(process.env.GOOGLE_SHEET_ID);
  await doc.useServiceAccountAuth({
    client_email: process.env.GOOGLE_SERVICE_EMAIL,
    private_key: process.env.GOOGLE_PRIVATE_KEY,
  });
  
  await doc.loadInfo();
  const sheet = doc.sheetsByIndex[0];
  const rows = await sheet.getRows();
  
  // Filter validated leads by campaign and date
  return rows.filter(row => {
    const leadDate = new Date(row.date);
    const campaignMatch = campaignIds.some(id => 
      row.campaign_id === id || row.campaign_name.includes(id)
    );
    
    return campaignMatch && 
           leadDate >= new Date(dateRange.start) && 
           leadDate <= new Date(dateRange.end) &&
           row.validated === 'true';
  });
}
```

## 🐛 Error Handling

```javascript
class FacebookAPIError extends Error {
  constructor(message, code, type) {
    super(message);
    this.code = code;
    this.type = type;
  }
}

class ReportGenerationError extends Error {
  constructor(message, adAccountId) {
    super(message);
    this.adAccountId = adAccountId;
  }
}

// Global error handler
app.use((error, req, res, next) => {
  console.error('Error:', error);
  
  if (error instanceof FacebookAPIError) {
    return res.status(400).json({
      success: false,
      error: 'Facebook API error',
      message: error.message,
      code: error.code
    });
  }
  
  if (error instanceof ReportGenerationError) {
    return res.status(500).json({
      success: false,
      error: 'Report generation failed',
      message: error.message,
      adAccountId: error.adAccountId
    });
  }
  
  res.status(500).json({
    success: false,
    error: 'Internal server error',
    message: 'Something went wrong'
  });
});
```
